import { motion } from "framer-motion";

import { TiltCard } from "./TiltCard";

/** The 3-bucket system as three fill tanks: monthly surplus drops in,
    Bucket 1 fills first, the rest overflows onward. Seed data (P8). */
export function BucketSchematic() {
  const reduce = usePrefersReducedMotion();
  const buckets = BLUEPRINT_SEED.buckets;

  return (
    <TiltCard>
      <div aria-label="3-bucket system">
        <p className="num text-[10px] tracking-[0.24em] text-dim">3-BUCKET SYSTEM</p>
        <p className="num mt-1 text-[10px] text-dim">↓ monthly surplus fills bucket 1 first</p>
        <div className="mt-4 flex items-end gap-3">
          {buckets.map((b, i) => (
            <div key={b.key} className="flex flex-1 flex-col items-center gap-2">
              <div className="relative h-28 w-full overflow-hidden rounded border border-line bg-void">
                <motion.div
                  className="absolute inset-x-0 bottom-0"
                  style={{
                    height: `${b.fillPct}%`,
                    transformOrigin: "bottom",
                    background: "var(--grad-wealth)",
                    opacity: 0.85,
                  }}
                  initial={{ scaleY: reduce ? 1 : 0 }}
                  animate={{ scaleY: 1 }}
                  transition={{ duration: 0.9, delay: i * 0.18, ease: [0.16, 1, 0.3, 1] }}
                />
                <span className="num absolute inset-x-0 top-1 text-center text-[10px] text-bone">
                  {b.fillPct}%
                </span>
              </div>
              <p className="num text-[9px] tracking-[0.14em] text-dim">BUCKET {i + 1}</p>
              <p className="text-center text-[12px] font-semibold text-bone">{b.name}</p>
              <p className="num text-center text-[9px] text-dim">{b.horizon}</p>
              <p className="text-center text-[9px] text-dim">{b.note}</p>
            </div>
          ))}
        </div>
        <p className="num mt-3 text-[10px] text-dim">
          bucket 1 ▸ bucket 2 ▸ bucket 3 — surplus overflows onward
        </p>
      </div>
    </TiltCard>
  );
}
