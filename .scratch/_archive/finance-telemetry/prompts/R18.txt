"use client";

import { motion } from "framer-motion";
import { usePrefersReducedMotion } from "@/app/lib/usePrefersReducedMotion";

type Item = { id: string; label: string };

/** The left-rail nav, styled as a speedo/steering-wheel cluster: an arc
    "wheel edge", one node per tab, a needle that springs to the active
    node, and two pulse rings radiating from it. Drives the parent tab. */
export function SpeedoNav({
  items,
  tab,
  onSelect,
}: {
  items: Item[];
  tab: string;
  onSelect: (id: string) => void;
}) {
  const reduce = usePrefersReducedMotion();
  const NODE_Y = [50, 150, 250, 350, 440];
  const active = Math.max(0, items.findIndex((it) => it.id === tab));
  const activeY = NODE_Y[Math.min(active, NODE_Y.length - 1)];

  return (
    <nav aria-label="Finance sections" className="speedo-nav w-16 shrink-0">
      <svg viewBox="0 0 64 480" className="h-full w-full" role="presentation">
        <path d="M 10 24 Q 2 240 10 456" fill="none" stroke="var(--sumi-line)" strokeWidth={2} />

        {!reduce && (
          <>
            <motion.circle
              cx={26}
              cy={activeY}
              r={5}
              fill="none"
              stroke="var(--amber)"
              strokeWidth={1}
              animate={{ r: [5, 17], opacity: [0.55, 0] }}
              transition={{ duration: 2, repeat: Infinity, ease: "easeOut" as const }}
            />
            <motion.circle
              cx={26}
              cy={activeY}
              r={5}
              fill="none"
              stroke="var(--amber)"
              strokeWidth={1}
              animate={{ r: [5, 17], opacity: [0.55, 0] }}
              transition={{ duration: 2, repeat: Infinity, ease: "easeOut" as const, delay: 1 }}
            />
          </>
        )}

        <motion.line
          x1={10}
          y1={240}
          x2={26}
          stroke="var(--amber)"
          strokeWidth={2}
          strokeLinecap="round"
          initial={{ y2: reduce ? activeY : NODE_Y[0] }}
          animate={{ y2: activeY }}
          transition={reduce ? { duration: 0 } : { type: "spring", stiffness: 120, damping: 18 }}
        />

        {items.map((it, i) => {
          const y = NODE_Y[Math.min(i, NODE_Y.length - 1)];
          const on = i === active;
          return (
            <g key={it.id} className="checkered-sweep" style={{ cursor: "pointer" }} onClick={() => onSelect(it.id)}>
              <rect x={0} y={y - 30} width={64} height={60} fill="transparent" />
              <circle
                cx={26}
                cy={y}
                r={5}
                fill={on ? "var(--amber)" : "var(--ink-void)"}
                stroke={on ? "var(--amber)" : "var(--sumi-line)"}
                strokeWidth={1.5}
              />
              <line
                x1={34}
                y1={y}
                x2={on ? 46 : 42}
                y2={y}
                stroke={on ? "var(--amber)" : "var(--sumi-line)"}
                strokeWidth={1.5}
              />
              <text
                transform={`rotate(-90 52 ${y})`}
                x={52}
                y={y}
                textAnchor="middle"
                dominantBaseline="middle"
                fontSize={7}
                letterSpacing={1}
                fill={on ? "var(--gold)" : "var(--bone-dim)"}
                style={{ fontFamily: "var(--font-data)" }}
              >
                {it.label}
              </text>
            </g>
          );
        })}
      </svg>
    </nav>
  );
}
