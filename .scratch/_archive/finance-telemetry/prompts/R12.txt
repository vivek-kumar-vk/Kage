"use client";

import { BLUEPRINT_SEED } from "@/app/lib/blueprintSeed";
import { useCountUp } from "@/app/lib/useCountUp";
import { formatINR } from "@/app/lib/formatINR";
import { motion } from "framer-motion";
import { usePrefersReducedMotion } from "@/app/lib/usePrefersReducedMotion";
import { TiltCard } from "./TiltCard";

/** Cash flow — income vs expenses, two bars drawing in, net surplus
    counting up. Seed data (P8). */
export function CashFlowPanel() {
  const reduce = usePrefersReducedMotion();
  const { income, expenses } = BLUEPRINT_SEED.cashFlow;
  const net = income - expenses;
  const shownNet = useCountUp(net);
  const rows = [
    { label: "INCOME", value: income, grad: true },
    { label: "EXPENSES", value: expenses, grad: false },
  ];
  const max = Math.max(income, expenses);

  return (
    <TiltCard>
      <div aria-label="Cash flow">
        <div className="flex items-baseline justify-between">
          <p className="num text-[10px] tracking-[0.24em] text-dim">CASH FLOW</p>
          <p className="num text-sm grad-text-wealth neon-glow-amber">
            net {formatINR(Math.round(shownNet))}
          </p>
        </div>
        {rows.map((r, i) => (
          <div key={r.label} className="mt-3">
            <div className="flex justify-between text-[11px]">
              <span className="text-dim">{r.label}</span>
              <span className="num text-bone">{formatINR(r.value)}</span>
            </div>
            <div className="mt-1 h-2 w-full overflow-hidden rounded bg-void">
              <motion.div
                className={`h-full ${r.grad ? "" : "bg-dim/60"}`}
                style={{
                  width: `${(r.value / max) * 100}%`,
                  transformOrigin: "left",
                  background: r.grad ? "var(--grad-wealth)" : undefined,
                }}
                initial={{ scaleX: reduce ? 1 : 0 }}
                animate={{ scaleX: 1 }}
                transition={{ duration: 0.9, delay: i * 0.15, ease: [0.16, 1, 0.3, 1] }}
              />
            </div>
          </div>
        ))}
      </div>
    </TiltCard>
  );
}
