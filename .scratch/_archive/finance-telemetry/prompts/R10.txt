"use client";

import { BLUEPRINT_SEED } from "@/app/lib/blueprintSeed";
import { useCountUp } from "@/app/lib/useCountUp";
import { formatINR } from "@/app/lib/formatINR";
import { motion } from "framer-motion";
import { usePrefersReducedMotion } from "@/app/lib/usePrefersReducedMotion";
import { TiltCard } from "./TiltCard";

/** Emergency-fund tiers (Instant / Core / Buffer) as one segmented
    bar-gauge that fills tier by tier toward the target. Seed data (P8). */
export function FundTiers() {
  const reduce = usePrefersReducedMotion();
  const { target, tiers } = BLUEPRINT_SEED.emergencyFund;
  const shown = useCountUp(target);
  const total = tiers.reduce((s, t) => s + t.amount, 0);
  const SHADE = ["bg-amber", "bg-amber/70", "bg-amber/40"];

  return (
    <TiltCard>
      <div aria-label="Emergency fund tiers">
        <div className="flex items-baseline justify-between">
          <p className="num text-[10px] tracking-[0.24em] text-dim">EMERGENCY FUND</p>
          <p className="num text-sm grad-text-wealth neon-glow-amber">
            {formatINR(Math.round(shown))} <span className="text-dim">/ {formatINR(target)}</span>
          </p>
        </div>
        <div className="mt-4 flex h-3 w-full gap-[2px] overflow-hidden rounded bg-void">
          {tiers.map((t, i) => (
            <motion.div
              key={t.key}
              className={`h-full ${SHADE[i]}`}
              style={{ width: `${(t.amount / total) * 100}%`, transformOrigin: "left" }}
              initial={{ scaleX: reduce ? 1 : 0 }}
              animate={{ scaleX: 1 }}
              transition={{ duration: 0.7, delay: i * 0.15, ease: [0.16, 1, 0.3, 1] }}
            />
          ))}
        </div>
        <div>
          {tiers.map((t, i) => (
            <div key={t.key} className="mt-2 flex items-center gap-2 text-[11px]">
              <span className={`inline-block h-2 w-2 rounded-sm ${SHADE[i]}`} />
              <span className="text-bone">{t.label}</span>
              <span className="num ml-auto text-dim">{formatINR(t.amount)}</span>
              <span className="text-dim">· {t.park}</span>
            </div>
          ))}
        </div>
      </div>
    </TiltCard>
  );
}
