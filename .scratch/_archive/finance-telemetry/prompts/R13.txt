"use client";

import { BLUEPRINT_SEED } from "@/app/lib/blueprintSeed";
import { motion } from "framer-motion";
import { usePrefersReducedMotion } from "@/app/lib/usePrefersReducedMotion";
import { TiltCard } from "./TiltCard";

/** Goals progress as RPM dials — a needle sweeps to each goal's percent
    and settles with a small wobble. Seed data (P8). */
export function GoalsGauges() {
  const reduce = usePrefersReducedMotion();
  const goals = BLUEPRINT_SEED.goals;

  const angleFor = (pct: number) => -135 + (pct / 100) * 270;
  const TRACK = "M 23.13 81.87 A 38 38 0 1 1 76.87 81.87";

  return (
    <TiltCard>
      <div aria-label="Goals progress">
        <p className="num text-[10px] tracking-[0.24em] text-dim">GOALS PROGRESS</p>
        <div className="mt-3 grid grid-cols-2 gap-4">
          {goals.map((g) => (
            <div key={g.label} className="flex flex-col items-center">
              <svg viewBox="0 0 100 100" className="h-24 w-24 neon-glow-amber">
                <path d={TRACK} fill="none" stroke="var(--sumi-line)" strokeWidth={6} strokeLinecap="round" />
                <path d={TRACK} fill="none" stroke="var(--gold)" strokeWidth={6} strokeLinecap="round"
                  pathLength={1}
                  style={{ strokeDasharray: 1, strokeDashoffset: 1 - g.pct / 100, opacity: 0.9 }} />
                <motion.g
                  style={{ transformOrigin: "50px 55px" }}
                  initial={{ rotate: reduce ? angleFor(g.pct) : -135 }}
                  animate={{ rotate: angleFor(g.pct) }}
                  transition={reduce ? { duration: 0 } : { type: "spring", stiffness: 60, damping: 12 }}
                >
                  <line x1={50} y1={55} x2={50} y2={25} stroke="var(--amber)" strokeWidth={3} strokeLinecap="round" />
                </motion.g>
                <circle cx={50} cy={55} r={4} fill="var(--amber)" />
                <text x={50} y={70} textAnchor="middle" className="num" fill="var(--bone)" fontSize={14}>
                  {g.pct}%
                </text>
              </svg>
              <p className="mt-1 text-center text-[11px] text-bone">{g.label}</p>
            </div>
          ))}
        </div>
      </div>
    </TiltCard>
  );
}
